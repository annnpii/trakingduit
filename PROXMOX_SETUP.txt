# ============================================================================
# 6. NGINX REVERSE PROXY (Optional, Recommended)
# ============================================================================

# Buat Nginx config
sudo nano /etc/nginx/sites-available/trakingduit

# Isi config (tanpa SSL dulu):
# ----------------------------------------------------------------
server {
    listen 80;
    server_name trakingduit.yourdomain.com;  # Ganti dengan domain lu

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;

    location / {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
        
        # Timeout settings
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }

    # Static files caching
    location /_next/static {
        proxy_pass http://localhost:3000;
        proxy_cache_valid 200 365d;
        add_header Cache-Control "public, immutable";
    }

    # Gzip compression
    gzip on;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml application/xml+rss text/javascript;
}
# ----------------------------------------------------------------

# Enable site
sudo ln -s /etc/nginx/sites-available/trakingduit /etc/nginx/sites-enabled/

# Test Nginx config
sudo nginx -t

# Restart Nginx
sudo systemctl restart nginx

# Test: http://your-server-ip atau http://yourdomain.com


# ============================================================================
# 7. SSL/HTTPS dengan Let's Encrypt (Optional, Recommended)
# ============================================================================

# Install SSL certificate
sudo certbot --nginx -d trakingduit.yourdomain.com

# Follow prompts:
# - Enter email
# - Agree to terms
# - Choose redirect HTTP to HTTPS (recommended)

# Auto-renewal test
sudo certbot renew --dry-run

# Certbot auto-renew sudah aktif via systemd timer
sudo systemctl status certbot.timer


# ============================================================================
# 8. FIREWALL SETUP (UFW)
# ============================================================================

# Enable firewall
sudo ufw enable

# Allow SSH (IMPORTANT! Jangan lupa ini)
sudo ufw allow 22/tcp

# Allow HTTP & HTTPS
sudo ufw allow 80/tcp
sudo ufw allow 443/tcp

# (Optional) Allow direct access ke port 3000 untuk testing
sudo ufw allow 3000/tcp

# Check firewall status
sudo ufw status verbose


# ============================================================================
# 9. PROXMOX SPECIFIC SETUP (Optional)
# ============================================================================

# Kalau pakai LXC Container di Proxmox:
# -----------------------------------------
# 1. Buat LXC container baru:
#    - Template: Ubuntu 22.04 atau Debian 11
#    - RAM: 2GB minimum (4GB recommended)
#    - Storage: 8GB minimum
#    - Network: Bridge ke interface utama
#    - Start on boot: Enable

# 2. Di Proxmox host, forward port (kalau perlu):
#    Edit /etc/pve/firewall/cluster.fw
#    Atau pakai iptables:
sudo iptables -t nat -A PREROUTING -i vmbr0 -p tcp --dport 80 -j DNAT --to 192.168.1.100:80
sudo iptables -t nat -A PREROUTING -i vmbr0 -p tcp --dport 443 -j DNAT --to 192.168.1.100:443

# 3. Backup container secara reguler:
#    Proxmox Web UI → Backup → Schedule
